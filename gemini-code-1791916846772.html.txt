    <script>
        const verbs = [
            // AAA
            { inf: "cut", past: "cut", part: "cut", group: "aaa", color: "var(--aaa-color)" },
            { inf: "hit", past: "hit", part: "hit", group: "aaa", color: "var(--aaa-color)" },
            { inf: "put", past: "put", part: "put", group: "aaa", color: "var(--aaa-color)" },
            // Ought/Aught
            { inf: "buy", past: "bought", part: "bought", group: "ought", color: "var(--ought-color)" },
            { inf: "catch", past: "caught", part: "caught", group: "ought", color: "var(--ought-color)" },
            { inf: "teach", past: "taught", part: "taught", group: "ought", color: "var(--ought-color)" },
            // I-A-U
            { inf: "sing", past: "sang", part: "sung", group: "iau", color: "var(--iau-color)" },
            { inf: "swim", past: "swam", part: "swum", group: "iau", color: "var(--iau-color)" },
            { inf: "ring", past: "rang", part: "rung", group: "iau", color: "var(--iau-color)" },
            // -EN
            { inf: "break", past: "broke", part: "broken", group: "en", color: "var(--en-color)" },
            { inf: "speak", past: "spoke", part: "spoken", group: "en", color: "var(--en-color)" },
            { inf: "drive", past: "drove", part: "driven", group: "en", color: "var(--en-color)" },
            // Misc
            { inf: "go", past: "went", part: "gone", group: "misc", color: "var(--misc-color)" },
            { inf: "see", past: "saw", part: "seen", group: "misc", color: "var(--misc-color)" },
            { inf: "make", past: "made", part: "made", group: "misc", color: "var(--misc-color)" }
        ];

        const grid = document.getElementById('verbGrid');
        const modal = document.getElementById('modal');

        verbs.forEach((verb, index) => {
            const box = document.createElement('div');
            box.className = 'verb-box';
            box.style.color = verb.color;
            box.innerHTML = `
                <div class="verb-num">${index + 1}</div>
                <div class="verb-inf">${verb.inf}</div>
            `;
            box.onclick = () => openModal(verb);
            grid.appendChild(box);
        });

        function openModal(verb) {
            document.getElementById('modalInf').innerText = verb.inf.toUpperCase();
            document.getElementById('modalInf').style.color = verb.color;
            document.getElementById('modalPast').innerText = verb.past;
            document.getElementById('modalPart').innerText = verb.part;
            document.getElementById('modalContent').style.borderColor = verb.color;
            modal.style.display = 'flex';
        }

        function closeModal() {
            modal.style.display = 'none';
        }

        // Cerrar al hacer clic fuera del modal
        window.onclick = function(event) {
            if (event.target == modal) {
                closeModal();
            }
        }
    </script>
